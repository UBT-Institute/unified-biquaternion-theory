\begin{abstract}
Assume the UBT covariant derivative is representation-preserving on a
complexified Lorentz-vector carrier $W_{\rm L}\simeq\mathbb C^4$, as in an
ordinary vector-bundle connection, and that an internal gauge group must
commute with local Lorentz transformations. The Lorentz commutant of one copy
of $W_{\rm L}$ is scalar. For a multiplicity space $M\simeq\mathbb C^m$,
\[
 \operatorname{End}_{\rm Lor}(W_{\rm L}\otimes M)
 \cong \mathbf1_{W_{\rm L}}\otimes\operatorname{End}(M).
\]
Thus the first multiplicity capable of supporting a faithful fundamental
$SU(3)$ internal action is $m=3$. Exact centralizer dimensions for $m=1,2,3$
are $1,4,9$ over $\mathbb C$. This converts the colour problem from searching
for $SU(3)$ inside the Lorentz biquaternion index to deriving a threefold
Lorentz-scalar degeneracy from the single UBT field or its constrained
spectrum. The existing single $S^1_\psi$ winding spectrum does not provide
that degeneracy.
\end{abstract}

\[

\]
For an ordinary connection on a vector bundle, $D_\mu$ preserves the fibre
representation: a section and its covariant derivative carry the same local
fibre representation, with the additional spacetime covector index on

The theorem below applies to any UBT branch in which:
\begin{enumerate}
 \item $D_\mu$ is a standard representation-preserving connection;

\section{Centralizer theorem}

\begin{theorem}[Lorentz multiplicity centralizer]
Let $W_{\rm L}$ be the complexified Lorentz-vector representation and let
$M\simeq\mathbb C^m$ carry no Lorentz action. On
\[
 W_{\rm tot}=W_{\rm L}\otimes M
\]
let Lorentz act as
\[
 \rho_{\rm L}(g)\otimes\mathbf1_M.
\]
Then
\[
 \boxed{
 \operatorname{End}_{\rm Lor}(W_{\rm tot})
 =\mathbf1_{W_{\rm L}}\otimes\operatorname{End}_{\mathbb C}(M).
 }
\]
In particular,
\[
 \dim_{\mathbb C}\operatorname{End}_{\rm Lor}(W_{\rm tot})=m^2.
\]
\end{theorem}

\begin{proof}
The complexified Lorentz vector representation is the irreducible
$(\mathbf2,\mathbf2)$ representation of
$\mathfrak{sl}_2(\mathbb C)\oplus\mathfrak{sl}_2(\mathbb C)$.
By Schur's lemma its endomorphism commutant on $W_{\rm L}$ is
$\mathbb C\mathbf1$. Tensoring with a trivial multiplicity space gives
exactly $\mathbf1\otimes\operatorname{End}(M)$.
For the explicit UBT matrix convention, the centralizer dimensions for
$m=1,2,3$ are independently verified by exact symbolic linear algebra in
\texttt{verification/su3\_lorentz\_multiplicity\_check.py}.
\end{proof}

\begin{corollary}[Minimal multiplicity for internal $SU(3)$]
A faithful fundamental internal $SU(3)$ commuting with Lorentz requires
\[
 \boxed{\dim_{\mathbb C}M\ge3.}
\]
For $m=1$ the commutant is only $\mathbb C$. For $m=2$ it is
$M_2(\mathbb C)$ and cannot contain a faithful copy of
$\mathfrak{sl}_3(\mathbb C)$. For $m=3$ the commutant is
$M_3(\mathbb C)$; after choosing an internal positive Hermitian form and
complex volume form, their common unitary stabiliser is $SU(3)$.
\end{corollary}
